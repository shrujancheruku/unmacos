#!/bin/bash
# UnMacOS: macOS settings for a tiling desktop. Re-runs when this file changes, which includes
# the Dock Settings below (dock_autohide, dock_size): changing one in your Answers re-runs it.
# Dock: autohide true, size 64
set -euo pipefail
echo "==> macOS settings"

# Dock: hidden, no recents; Spaces stay in order (Rift needs this)
defaults write com.apple.dock autohide -bool true
defaults write com.apple.dock autohide-delay -float 0
defaults write com.apple.dock tilesize -int 64
defaults write com.apple.dock show-recents -bool false
defaults write com.apple.dock mru-spaces -bool false
defaults write com.apple.dock expose-group-apps -bool true

# Displays have separate Spaces (required by Rift)
defaults write com.apple.spaces spans-displays -bool false

# Hide the menu bar (SketchyBar replaces it); dark mode
defaults write NSGlobalDomain _HIHideMenuBar -bool true
defaults write NSGlobalDomain AppleInterfaceStyle -string Dark

# Keyboard: hold a key to repeat it (no accent popup)
defaults write NSGlobalDomain ApplePressAndHoldEnabled -bool false

# Ctrl+Cmd drag to move windows from anywhere
defaults write NSGlobalDomain NSWindowShouldDragOnGesture -bool true

# Stage Manager / native tiling off (Rift does tiling)
defaults write com.apple.WindowManager GloballyEnabled -bool false
defaults write com.apple.WindowManager EnableStandardClickToShowDesktop -bool false
defaults write com.apple.WindowManager EnableTilingByEdgeDrag -bool false
defaults write com.apple.WindowManager EnableTopTilingByEdgeDrag -bool false

# Finder: list view, path bar, extensions
defaults write com.apple.finder FXPreferredViewStyle -string Nlsv
defaults write com.apple.finder ShowPathbar -bool true
defaults write NSGlobalDomain AppleShowAllExtensions -bool true

echo "Log out and back in for Spaces and menu bar changes."

=== .chezmoiscripts/40-no-ai-no-glass.sh
#!/bin/bash
# UnMacOS: on macOS 26+ (Tahoe, Golden Gate), turn off Apple Intelligence and
# Liquid Glass to get as close to a Sequoia desktop as macOS allows.
# Re-runs when this script, the profile or the macOS version changes (upgrades
# reset these settings).
# profile hash: <hash>
# macOS: <macos>
#
# Opt out: UNMACOS_KEEP_AI=1 / UNMACOS_KEEP_GLASS=1 chezmoi apply
# Undo:    remove the "UnMacOS: no Apple Intelligence" profile in Device Management,
#          defaults delete -g com.apple.SwiftUI.DisableSolarium,
#          defaults delete -g NSConvolutionOverride1 (and NSConvolutionOverride2),
#          sudo rm /Library/Preferences/FeatureFlags/Domain/WritingTools.plist
set -euo pipefail
# Your unmacos-setup choices: {"accent":false,"ai":true,"firefox":true,"firefox_wallpaper":false,"glass":false,"orca":false,"spotify":true,"wallpaper":true}
# shellcheck source=/dev/null # prefs.sh is generated by chezmoi on the Mac, not in the repo
[[ -r "$HOME/.config/unmacos/prefs.sh" ]] && . "$HOME/.config/unmacos/prefs.sh"

major="$(sw_vers -productVersion | cut -d. -f1)"
if (( major < 26 )); then
  echo "==> macOS $major: no Apple Intelligence / Liquid Glass changes needed"
  # Sequoia: still opt out of the Apple Intelligence waitlist/prompt
  defaults write com.apple.CloudSubscriptionFeatures.optIn "545129924" -bool false 2>/dev/null || true
  exit 0
fi

if [[ "${UNMACOS_KEEP_AI:-}" != 1 ]]; then
  echo "==> Apple Intelligence off"
  profile="$HOME/.local/share/unmacos/profiles/unmacos-no-ai.mobileconfig"
  if profiles show -type configuration 2>/dev/null | grep -q "io.github.unmacos.no-ai"; then
    echo "    profile already installed"
  else
    # macOS needs a person to approve profiles; this queues it in System Settings
    open "$profile"
    sleep 2
    open "x-apple.systempreferences:com.apple.Profiles-Settings.extension" || true
    echo "    Approve 'UnMacOS: no Apple Intelligence' in System Settings > General > Device Management"
  fi

  # Writing Tools popup in every text field (needs sudo, so only when it isn't off yet)
  if ! defaults read /Library/Preferences/FeatureFlags/Domain/WritingTools LightweightUI_macOS 2>/dev/null | grep -q "Enabled = 0"; then
    sudo mkdir -p /Library/Preferences/FeatureFlags/Domain
    sudo defaults write /Library/Preferences/FeatureFlags/Domain/WritingTools LightweightUI_macOS -dict Enabled -bool false
  fi

  # Siri, inline predictions and summaries (the profile enforces these; set them too)
  defaults write com.apple.assistant.support "Assistant Enabled" -bool false
  defaults write com.apple.Siri StatusMenuVisible -bool false
  defaults write com.apple.Siri VoiceTriggerUserEnabled -bool false
  defaults write NSGlobalDomain NSAutomaticInlinePredictionEnabled -bool false
  defaults write com.apple.CloudSubscriptionFeatures.optIn "545129924" -bool false 2>/dev/null || true
fi

if [[ "${UNMACOS_KEEP_GLASS:-}" != 1 ]]; then
  echo "==> Liquid Glass off"
  # Solid backgrounds instead of glass (may need Full Disk Access for the terminal)
  if ! defaults write com.apple.universalaccess reduceTransparency -bool true 2>/dev/null; then
    echo "    Could not set Reduce Transparency: turn it on in"
    echo "    System Settings > Accessibility > Display"
  fi
  # Pre-Liquid Glass look for SwiftUI apps; Control Center keeps the new UI so
  # right-side menu bar items keep working
  defaults write -g com.apple.SwiftUI.DisableSolarium -bool true
  defaults write com.apple.ControlCenter com.apple.SwiftUI.DisableSolarium -bool false
  # Small window corners (private AppKit key; 1 = windows/sheets, 2 = panels), the
  # same radius as the bar's chips: corner_radius in setup.toml. Apps pick it up when
  # they relaunch.
  defaults write -g NSConvolutionOverride1 -float 4
  defaults write -g NSConvolutionOverride2 -float 4
  if (( major >= 27 )); then
    echo "    Also drag System Settings > Appearance > Liquid Glass to the minimum (no setting for scripts)"
    open "x-apple.systempreferences:com.apple.Appearance-Settings.extension" || true
  fi
fi

echo "Log out and back in to finish (Liquid Glass changes and Writing Tools need it)."

=== .chezmoiscripts/80-firefox.sh
#!/bin/bash
# UnMacOS: Firefox follows the UnMacOS theme through Pywalfox.
#   - pywalfox (uv tool) + its native messaging host, so the extension can read
#     the colors unmacos-theme-render writes to ~/.cache/wal/colors.json
#   - a Firefox policy that installs the Pywalfox and Dark Reader extensions
#     (no "Add" click); Dark Reader recolors pages from the rendered darkreader.json
# Opt out: UNMACOS_KEEP_FIREFOX=1 chezmoi apply
# Undo:    uv tool uninstall pywalfox,
#          rm ~/Library/Application\ Support/Mozilla/NativeMessagingHosts/pywalfox.json,
#          defaults delete org.mozilla.firefox ExtensionSettings,
#          then remove Pywalfox and Dark Reader in Firefox > Add-ons
set -euo pipefail
# Your unmacos-setup choices: {"accent":false,"ai":true,"firefox":true,"firefox_wallpaper":false,"glass":false,"orca":false,"spotify":true,"wallpaper":true}
# shellcheck source=/dev/null # prefs.sh is generated by chezmoi on the Mac, not in the repo
[[ -r "$HOME/.config/unmacos/prefs.sh" ]] && . "$HOME/.config/unmacos/prefs.sh"
[[ "${UNMACOS_KEEP_FIREFOX:-}" == 1 ]] && exit 0
eval "$(/opt/homebrew/bin/brew shellenv)"

echo "==> Firefox theme (Pywalfox)"
uv tool install --quiet pywalfox
# macOS 27 blocks the Firefox-launched daemon from reading Firefox's
# profiles.ini, so hand it a copy outside the Firefox folder that points at the
# profile Firefox actually uses.
ff="$HOME/Library/Application Support/Firefox"
pf="$HOME/.local/state/unmacos/firefox"
profile="$(awk -F= '/^\[Install/ {i=1} i && $1 == "Default" {print $2; exit}' "$ff/profiles.ini" 2>/dev/null || true)"
args=()
if [[ -n "$profile" ]]; then
  mkdir -p "$pf"
  printf '[Profile0]\nName=unmacos\nIsRelative=0\nPath=%s\n' "$ff/$profile" > "$pf/profiles.ini"
  args=(--profile-path "$pf")
else
  echo "    No Firefox profile yet: open Firefox once, then run chezmoi apply again"
fi
"$HOME/.local/bin/pywalfox" install --executable "$HOME/.local/share/unmacos/pywalfox-host" "${args[@]}" >/dev/null

defaults write org.mozilla.firefox EnterprisePoliciesEnabled -bool true
defaults write org.mozilla.firefox ExtensionSettings -dict-add pywalfox@frewacom.org \
  '<dict><key>installation_mode</key><string>normal_installed</string><key>install_url</key><string>https://addons.mozilla.org/firefox/downloads/latest/pywalfox/latest.xpi</string></dict>'
# Dark Reader recolors web pages; unmacos-theme-render writes darkreader.json to import
defaults write org.mozilla.firefox ExtensionSettings -dict-add addon@darkreader.org \
  '<dict><key>installation_mode</key><string>normal_installed</string><key>install_url</key><string>https://addons.mozilla.org/firefox/downloads/latest/darkreader/latest.xpi</string></dict>'

# Colors for the current theme (the renderer links them on every theme switch)
"$HOME/.local/bin/unmacos-theme-render" >/dev/null
# New tab wallpaper = desktop wallpaper (unmacos-wallpaper keeps it in step)
# (it also installs the live-update autoconfig into Firefox.app, and says if Firefox needs a restart)
"$HOME/.local/bin/unmacos-firefox-wallpaper" 2>/dev/null | sed "s/^unmacos-firefox-wallpaper: /    /" || true
echo "    Restart Firefox, then Pywalfox icon > Fetch Pywal colors (once); theme switches then apply live."

=== .chezmoiscripts/85-spotify.sh
#!/bin/bash
# UnMacOS: the Spotify app follows the UnMacOS theme through Spicetify.
#   Points Spicetify at Themes/UnMacOS (color.ini links to the rendered
#   spicetify-color.ini) and patches Spotify. unmacos-theme-render then runs
#   recolors an open Spotify live on every theme switch, over a DevTools port
#   on 127.0.0.1:9222 (spotify_launch_flags; while it's open, local programs
#   running as you can drive Spotify's UI).
# Only runs when both Spotify.app and spicetify are installed (Brewfile.personal).
# Opt out: UNMACOS_KEEP_SPOTIFY=1 chezmoi apply
# Undo:    spicetify restore
set -euo pipefail
# Your unmacos-setup choices: {"accent":false,"ai":true,"firefox":true,"firefox_wallpaper":false,"glass":false,"orca":false,"spotify":true,"wallpaper":true}, corner radius 4
# shellcheck source=/dev/null # prefs.sh is generated by chezmoi on the Mac, not in the repo
[[ -r "$HOME/.config/unmacos/prefs.sh" ]] && . "$HOME/.config/unmacos/prefs.sh"
[[ "${UNMACOS_KEEP_SPOTIFY:-}" == 1 ]] && exit 0
eval "$(/opt/homebrew/bin/brew shellenv)"
command -v spicetify >/dev/null && [[ -d /Applications/Spotify.app ]] || exit 0

echo "==> Spotify theme (Spicetify)"
"$HOME/.local/bin/unmacos-theme-render" >/dev/null 2>&1 || true
# `--`: stop spicetify reading the --remote-debugging-port value as its own flag
spicetify -q config -- \
  spotify_path "/Applications/Spotify.app/Contents/Resources" \
  prefs_path "$HOME/Library/Application Support/Spotify/prefs" \
  current_theme UnMacOS color_scheme UnMacOS \
  inject_css 1 replace_colors 1 overwrite_assets 0 inject_theme_js 0 \
  spotify_launch_flags "--remote-debugging-port=9222"
# Spotify updates replace the patched files; a fresh backup is needed after one.
# -n: spicetify's restart quits Spotify but doesn't relaunch it on macOS, so
# reopen an open Spotify here instead (with the port, for live recoloring).
if ! spicetify -q -n backup apply 2>/dev/null; then
  spicetify -q -n restore backup apply
fi
if pgrep -xq Spotify; then
  osascript -e 'tell application "Spotify" to quit' 2>/dev/null || true
  for _ in $(seq 40); do pgrep -xq Spotify || break; sleep 0.25; done
  pgrep -xq Spotify && { pkill -x Spotify; sleep 1; }
  "$HOME/.local/bin/unmacos-spotify" -g
fi

=== .chezmoiscripts/10-brew-bundle.sh
#!/bin/bash
# UnMacOS: install Homebrew packages. Re-runs when a Brewfile or your app choices change.
# Brewfile hash: <hash>
# App groups: dev media; skipped: vlc
# dev hash: <hash>
# media hash: <hash>
set -euo pipefail
if [[ ! -x /opt/homebrew/bin/brew ]]; then
  NONINTERACTIVE=1 /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
fi
eval "$(/opt/homebrew/bin/brew shellenv)"
echo "==> brew bundle (core)"
brew bundle --file="<repo>/Brewfile"

# The chosen groups in one Brewfile, without the apps you unticked
picked="$(mktemp -t unmacos-brewfile)"
trap 'rm -f "$picked"' EXIT
for g in "dev" "media"; do
  cat "<repo>/brew/$g.Brewfile"
done | awk -v skip="vlc" '
  BEGIN { n = split(skip, s, "|"); for (i = 1; i <= n; i++) drop[s[i]] = 1 }
  /^(brew|cask|mas) "/ { split($0, q, "\""); id = q[2]; sub(/.*\//, "", id); if (id in drop) next }
  { print }' > "$picked"
echo "==> brew bundle (dev, media)"
brew bundle --file="$picked" || echo "Some optional apps failed; continuing."
